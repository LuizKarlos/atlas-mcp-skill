import test from 'node:test';
import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import { mkdtempSync, writeFileSync, rmSync } from 'node:fs';

const SCRIPT = fileURLToPath(new URL('../scripts/wcag_check.js', import.meta.url));
const run = (args) => spawnSync(process.execPath, [SCRIPT, ...args], { encoding: 'utf8' });

test('wcag argv grammar: equivalence, unknown flags, separated values, help, and json', () => {
  // ponytail: minimal subprocess runner covering argv grammar contract
  assert.equal(checkWcag({ fg: '#000000', bg: '#ffffff' }).valid, true);

  const eq = run(['--fg=#000', '--bg=#fff', '--width=48', '--json']);
  const sep = run(['--fg', '#000', '--bg', '#fff', '--width', '48', '--json']);
  assert.equal(eq.status, 0);
  assert.equal(sep.status, 0);
  assert.deepEqual(JSON.parse(eq.stdout), JSON.parse(sep.stdout));

  for (const flag of [['--foreground=#fff'], ['--background', '#fff'], ['--typo'], ['-x']]) {
    assert.notEqual(run(flag).status, 0);
  }

  const noFile = run(['--fg', '#000', '--json']);
  assert.equal(noFile.status, 0);

  const bad = [['--fg'], ['--fg='], ['--fg', '""'], ['--fg', '--bg', '#fff'], ['a.json', 'b.json'], ['a.json', '--fg', '#000'], ['--fg', '#0', '--fg', '#1']];
  for (const b of bad) {
    assert.notEqual(run(b).status, 0);
  }

  const help = run(['--help']);
  const helpFile = run(['-h', 'missing.json']);
  assert.equal(help.status, 0);
  assert.equal(helpFile.status, 0);
  assert.match(help.stdout, /Usage:/);

  const tmpDir = mkdtempSync(`${tmpdir()}/wcag-`);
  const tmpFile = `${tmpDir}/test.json`;
  try {
    writeFileSync(tmpFile, JSON.stringify({ fg: '#000000', bg: '#ffffff', width: 44, height: 44 }));
    const fRes = run([tmpFile, '--json']);
    assert.equal(fRes.status, 0);
    assert.equal(JSON.parse(fRes.stdout).valid, true);
  } finally {
    rmSync(tmpDir, { recursive: true, force: true });
  }
});

tests/issue351_wcag_precision.test.js

import test from "node:test";
import assert from "node:assert/strict";
import { checkWcag, getContrastRatio } from "../scripts/wcag_check.js";

function oracleLum(hex) {
  const c = [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16) / 255);
  const lin = c.map((v) => (v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4));
  return 0.2126 * lin[0] + 0.7152 * lin[1] + 0.0722 * lin[2];
}

function oracleRatio(fg, bg) {
  const l1 = oracleLum(fg), l2 = oracleLum(bg);
  return (Math.max(l1, l2) + 0.05) / (Math.min(l1, l2) + 0.05);
}

test("caso #070707/#777777 reprova passAA e valid usando razão exata", () => {
  const res = checkWcag({ fg: "#070707", bg: "#777777" });
  const ratio = getContrastRatio("#070707", "#777777");
  const expectedRatio = oracleRatio("#070707", "#777777");

  assert.equal(ratio, expectedRatio);
  assert.ok(ratio < 4.5);
  assert.equal(res.contrast.ratio, expectedRatio);
  assert.equal(res.contrast.passAA, false);
  assert.equal(res.contrast.passLarge, true);
  assert.equal(res.valid, false);
});

test("oraculo sRGB independente nos limites 4.5 e 3.0 e preservacao de 21 e 1", () => {
  assert.equal(getContrastRatio("#000000", "#ffffff"), 21);
  assert.equal(oracleRatio("#000000", "#ffffff"), 21);
  assert.equal(getContrastRatio("#ffffff", "#ffffff"), 1);
  assert.equal(oracleRatio("#ffffff", "#ffffff"), 1);

  const r45Below = oracleRatio("#070707", "#777777");
  const r45Above = oracleRatio("#060606", "#777777");
  assert.ok(r45Below < 4.5 && r45Below >= 4.49);
  assert.ok(r45Above >= 4.5);
  assert.equal(checkWcag({ fg: "#070707", bg: "#777777" }).contrast.passAA, false);
  assert.equal(checkWcag({ fg: "#060606", bg: "#777777" }).contrast.passAA, true);

  const r30Below = oracleRatio("#2e2f2f", "#777777");
  const r30Above = oracleRatio("#2e2e2e", "#777777");
  assert.ok(r30Below < 3.0 && r30Below >= 2.99);
  assert.ok(r30Above >= 3.0);
  assert.equal(checkWcag({ fg: "#2e2f2f", bg: "#777777" }).contrast.passLarge, false);
  assert.equal(checkWcag({ fg: "#2e2e2e", bg: "#777777" }).contrast.passLarge, true);
});

import test from "node:test";
import assert from "node:assert/strict";
import http from "node:http";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";
import { execFile } from "node:child_process";

const BIN = path.resolve("bin/megabrain.js");

test("authenticated status and setup exit non-zero on 401 and 0 on valid token", async () => {
  const tmp = fs.mkdtempSync(path.join(os.tmpdir(), "s11-"));
  const server = http.createServer((req, res) => {
    if (req.url === "/health") {
      res.writeHead(200, { "content-type": "application/json" });
      return res.end(JSON.stringify({ status: "HEALTHY", ok: true }));
    }
    if (req.headers.authorization === "Bearer valid") {
      let b = "";
      req.on("data", (c) => (b += c));
      return req.on("end", () => {
        const id = JSON.parse(b || "{}").id ?? 1;
        res.writeHead(200, { "content-type": "application/json" });
        res.end(JSON.stringify({ jsonrpc: "2.0", id, result: { tools: [] } }));
      });
    }
    res.writeHead(401, { "content-type": "application/json" });
    res.end(JSON.stringify({ error: { message: "Unauthorized" } }));
  });
  await new Promise((r) => server.listen(0, "127.0.0.1", r));
  const endpoint = `http://127.0.0.1:${server.address().port}/mcp`;
  const run = (cmd, token) =>
    new Promise((resolve) => {
      execFile(
        process.execPath,
        [BIN, cmd, "--endpoint", endpoint, ...(cmd === "setup" ? ["--dest", tmp] : []), "--json"],
        { env: { ...process.env, XDG_CONFIG_HOME: tmp, MEGABRAIN_TOKEN: token } },
        (err) => resolve(err ? (err.code ?? 1) : 0)
      );
    });

  try {
    assert.notEqual(await run("status", "bad"), 0);
    assert.notEqual(await run("setup", "bad"), 0);
    assert.equal(await run("status", "valid"), 0);
    assert.equal(await run("setup", "valid"), 0);
  } finally {
    server.close();
    fs.rmSync(tmp, { recursive: true, force: true });
  }
});

  const canary = "canary-secret-token-xyz";
  const bads = [
    `http://127.0.0.1:8000/mcp?token=${canary}`,
    `http://user:${canary}@127.0.0.1:8000/mcp`,

  ];
  let fetchCalls = 0;
  const origFetch = globalThis.fetch;
  const logs = [];

  console.log = (...a) => logs.push(a.join(" "));

import test from 'node:test';

test('Stress: DAG Validator - 50 nodes wide parallel waves', () => {

test('Stress: DAG Validator - Complex cycles in 50 nodes', () => {
  // 50 nodes total ring cycle: 0 -> 1 -> ... -> 49 -> 0
  const ring = Array.from({ length: 50 }, (_, i) => ({
    id: `ring_${i}`,
    depends_on: [`ring_${(i + 49) % 50}`],
  }));
  const resRing = validateDAG(ring);
  assert.equal(resRing.valid, false);
  assert.equal(resRing.cycles.length, 50);
  assert.equal(resRing.waves.length, 0);

  // 45 acyclic nodes + 5 cyclic nodes
  const mixed = Array.from({ length: 45 }, (_, i) => ({
    id: `ok_${i}`,
    depends_on: i === 0 ? [] : [`ok_${i - 1}`],
  }));
  for (let i = 0; i < 5; i++) {
    mixed.push({
      id: `cycle_${i}`,
      depends_on: ['ok_44', `cycle_${(i + 4) % 5}`],
    });
  }
  const resMixed = validateDAG(mixed);
  assert.equal(resMixed.valid, false);
  assert.deepEqual(resMixed.cycles.sort(), ['cycle_0', 'cycle_1', 'cycle_2', 'cycle_3', 'cycle_4'].sort());
  assert.equal(resMixed.waves.length, 45);

  // Figure-8 double cycle: c1_0 <-> c1_1 and c2_0 <-> c2_1 sharing a hub
  const fig8 = [
    { id: 'hub', depends_on: ['c1_b', 'c2_b'] },
    { id: 'c1_a', depends_on: ['hub'] },
    { id: 'c1_b', depends_on: ['c1_a'] },
    { id: 'c2_a', depends_on: ['hub'] },
    { id: 'c2_b', depends_on: ['c2_a'] },
  ];
  const resFig8 = validateDAG(fig8);
  assert.equal(resFig8.valid, false);
  assert.equal(resFig8.cycles.length, 5);

  // Self-loop: node depends on itself
  const selfLoop = [
    { id: 'start', depends_on: [] },
    { id: 'loop', depends_on: ['start', 'loop'] },
  ];
  const resSelf = validateDAG(selfLoop);

test('Stress: Merkle Verify - Tampering intermediate block and hash alteration', () => {
  const blocks = [];
  for (let i = 0; i < 50; i++) {
    const b = {
      index: i,
      previous_hash: i === 0 ? '0' : blocks[i - 1].hash,
      data: { sequence: i, payload: `payload_${i}` },
    };
    b.hash = computeBlockHash(b);
    blocks.push(b);
  }

  // 1. Tamper intermediate payload at block 25 without altering declared hash
  const tamperedPayload = JSON.parse(JSON.stringify(blocks));
  tamperedPayload[25].data.payload = 'tampered_malicious_data';
  const res1 = verifyLedger(tamperedPayload);
  assert.equal(res1.valid, false);
  assert.ok(res1.errors.some((e) => e.includes('Block 25: hash mismatch')));

  // 2. Tamper block 25 data AND recalculate block 25 hash
  // Block 26 previous_hash will no longer match block 25's new hash
  const tamperedChain = JSON.parse(JSON.stringify(blocks));
  tamperedChain[25].data.payload = 'tampered_malicious_data';
  tamperedChain[25].hash = computeBlockHash(tamperedChain[25]);
  const res2 = verifyLedger(tamperedChain);
  assert.equal(res2.valid, false);
  assert.ok(res2.errors.some((e) => e.includes('Block 26: previous_hash mismatch')));

  // 3. Alter declared hash of block 30 directly
  const tamperedHash = JSON.parse(JSON.stringify(blocks));
  tamperedHash[30].hash = 'deadbeefdeadbeefdeadbeefdeadbeefdeadbeefdeadbeefdeadbeefdeadbeef';
  const res3 = verifyLedger(tamperedHash);
  assert.equal(res3.valid, false);
  assert.ok(res3.errors.some((e) => e.includes('Block 30: hash mismatch')));
  assert.ok(res3.errors.some((e) => e.includes('Block 31: previous_hash mismatch')));
});

    }
  }

test('Privacy & Neutrality Scanner', () => {
  const forbidden = [
    ['Luiz', 'Karlos'].join(''),
    ['Luiz', 'Karlos', '/megabrain'].join(''),
  ];
  const selfPath = fileURLToPath(import.meta.url);

  function scan(dir) {
    const hits = [];
    for (const entry of readdirSync(dir, { withFileTypes: true })) {
      if (entry.name === '.git') continue;
      const full = join(dir, entry.name);
      if (entry.isDirectory()) {
        hits.push(...scan(full));
      } else if (entry.isFile()) {
        if (full === selfPath) continue;
        const text = readFileSync(full, 'utf8');
        for (const term of forbidden) {
          if (text.includes(term)) {
            hits.push({ file: full, term });
          }
        }
      }
    }
    return hits;
  }

  const violations = scan(repoRoot);
  assert.deepEqual(violations, [], `Privacy violations found: ${JSON.stringify(violations)}`);
});
