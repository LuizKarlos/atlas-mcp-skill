export async function run({ positionals = [], flags = {} } = {}) {
  if (flags?.help) {
    console.log("Usage: megabrain setup [--dest <path>] [--endpoint <url>] [--json]");
    return;
  }

  const endpoint = flags?.endpoint || getEndpoint();
  validateEndpoint(endpoint);
  const token = getToken();
  const rootDir = fileURLToPath(new URL("../../", import.meta.url));
  const destArg = typeof flags?.dest === "string" ? flags.dest : (positionals[0] || null);
  const destDir = destArg || path.join(process.cwd(), ".agents", "skills");

  fs.mkdirSync(destDir, { recursive: true });
  const installed = [];
  for (const item of [...SKILLS, "scripts"]) {
    const src = path.join(rootDir, item);
    if (fs.existsSync(src)) {
      fs.cpSync(src, path.join(destDir, item), { recursive: true });
      installed.push(item);
    }
  }

  const harnesses = [];
  const cursorDir = path.join(process.cwd(), ".cursor");
  if (fs.existsSync(cursorDir)) {
    harnesses.push(".cursor");
    const rulesDir = path.join(cursorDir, "rules");
    fs.mkdirSync(rulesDir, { recursive: true });
    const rulePath = path.join(rulesDir, "megabrain.mdc");
    if (!fs.existsSync(rulePath)) fs.writeFileSync(rulePath, "# Atlas Rules\nUse Atlas MCP tools (atlas.*).\n");
  }
  if (fs.existsSync(path.join(process.cwd(), ".vscode"))) harnesses.push(".vscode");
  if (fs.existsSync(path.join(os.homedir(), ".hermes"))) harnesses.push(".hermes");

  const mcpConfig = {
    mcpServers: {
      atlas: {
        url: endpoint,
        headers: token ? { Authorization: "Bearer ${MEGABRAIN_TOKEN}" } : {}
      }
    }
  };
  if (flags?.["write-mcp"] || flags?.writeMcp) fs.writeFileSync(path.join(destDir, "megabrain.mcp.json"), JSON.stringify(mcpConfig, null, 2) + "\n");

  const nodeMajor = parseInt(process.versions.node.split(".")[0], 10);
  const nodeOk = nodeMajor >= 18;

  function healthReady(health) {
    const rawStatus = typeof health?.status === "string" ? health.status.toUpperCase() : "";
    return Boolean(health) && !health.isError && health.ready !== false && health.kernel_ready !== false && health.ok !== false &&
      (["READY", "HEALTHY", "OK"].includes(rawStatus) || (rawStatus === "" && (health.ready === true || health.ok === true)));
  }

  const explicitToken = flags?.token || process.env.MEGABRAIN_TOKEN;
  let connected = false;
  const start = Date.now();
  try {
    const health = await checkHealth({ endpoint, token: explicitToken, signal: AbortSignal.timeout(3000) });
    connected = healthReady(health);
    if (connected && explicitToken) await listTools({ endpoint, token: explicitToken, signal: AbortSignal.timeout(3000) });
  } catch {
    connected = false;
  }
  const latencyMs = Date.now() - start;

  const report = {
    status: nodeOk && connected ? "ok" : "degraded",
    prerequisites: { node: { version: process.version, valid: nodeOk } },
    connectivity: { endpoint, connected, latencyMs },
    installation: { destination: destDir, installed },
    harnesses,
    mcpConfig
  };

  if (flags?.json) {
    console.log(JSON.stringify(report));
  } else {
    console.log(`MegaBrain Setup: ${report.status.toUpperCase()}`);
    console.log(`- Node.js: ${process.version} (${nodeOk ? "OK" : "Requires >= 18"})`);
    console.log(`- Connectivity: ${connected ? "Connected" : "Offline"} (${latencyMs}ms) to ${endpoint}`);
    console.log(`- Installed ${installed.length} items to ${destDir}`);
    if (harnesses.length) console.log(`- Harnesses: ${harnesses.join(", ")}`);
  }
  if (report.status !== "ok") process.exitCode = 1;
  return report;
}

import { checkHealth, listTools } from "../client.js";

// ponytail: unary health ping; upgrade to full telemetry snapshot when dashboard ready
export async function run({ flags = {} }) {
  const endpoint = flags.endpoint || getEndpoint();
  const explicitToken = flags.token || process.env.MEGABRAIN_TOKEN;
  const health = await checkHealth({ endpoint, token: explicitToken });
  const rawStatus = typeof health?.status === "string" ? health.status.toUpperCase() : "";
  let isOk = Boolean(health) && !health.isError && health.ready !== false && health.kernel_ready !== false && health.ok !== false &&
    (["READY", "HEALTHY", "OK"].includes(rawStatus) || (rawStatus === "" && (health.ready === true || health.ok === true)));
  if (isOk && explicitToken) {
    try { await listTools({ endpoint, token: explicitToken }); } catch { isOk = false; }
  }
  if (!isOk) process.exitCode = 1;
  const status = isOk
    ? (health?.status || "ok")
    : (["ERROR", "DEGRADED", "NOT_READY"].includes(rawStatus) ? health.status : (health?.ready === false ? "not_ready" : "error"));
  const out = { endpoint, status, health };
  if (flags.json) {
    console.log(JSON.stringify(out));
  } else {
    console.log(`Endpoint: ${endpoint}\nStatus: ${out.status}`);
  }
  return out;
}

import fs from "node:fs";
import path from "node:path";
import os from "node:os";

// ponytail: sync fs operations for CLI config; upgrade to async if integrated into high-concurrency daemon
export function getConfigDir() {
  return process.env.XDG_CONFIG_HOME
    ? path.join(process.env.XDG_CONFIG_HOME, "megabrain")
    : path.join(os.homedir(), ".config", "megabrain");
}

export function getConfigFile() {
  return path.join(getConfigDir(), "credentials.json");
}

export function loadConfig() {
  try {
    const file = getConfigFile();
    if (!fs.existsSync(file)) return {};
    return JSON.parse(fs.readFileSync(file, "utf8"));
  } catch {
    return {};
  }
}

const DEFAULT_ENDPOINT = "https://mcp.atlascommand.dev/mcp";
const SENSITIVE_QUERY_KEY = /token|key|bearer/i;

// ponytail: rejeita esquemas nao http(s), userinfo e query sensivel; upgrade to allowlist se endpoints legit precisarem query.
export function validateEndpoint(endpoint) {
  if (typeof endpoint !== "string" || endpoint.trim() === "") {
    throw new Error("Endpoint invalido: valor ausente");
  }
  let url;
  try {
    url = new URL(endpoint);
  } catch {
    throw new Error("Endpoint invalido: URL malformada");
  }
  if (url.protocol !== "https:" && url.protocol !== "http:") {
    throw new Error(`Endpoint invalido: protocolo nao permitido (${url.protocol})`);
  }
  if (!url.hostname) throw new Error("Endpoint invalido: host ausente");
  if (url.username || url.password) {
    throw new Error("Endpoint invalido: credenciais (userinfo) nao permitidas");
  }
  for (const key of url.searchParams.keys()) {
    if (SENSITIVE_QUERY_KEY.test(key)) {
      throw new Error("Endpoint invalido: parametro de consulta sensivel nao permitido");
    }
  }
  return endpoint;
}

export function saveConfig(data) {
  const dir = getConfigDir();
  const file = getConfigFile();
  const current = loadConfig();
  const merged = { ...current, ...data };
  if (merged.endpoint !== undefined && merged.endpoint !== null) {
    merged.endpoint = validateEndpoint(merged.endpoint);
  }
  fs.mkdirSync(dir, { recursive: true, mode: 0o700 });
  fs.chmodSync(dir, 0o700);
  const tmp = `${file}.tmp.${process.pid}.${Date.now()}`;
  try {
    fs.writeFileSync(tmp, JSON.stringify(merged, null, 2) + "\n", { mode: 0o600 });
    fs.chmodSync(tmp, 0o600);
    fs.renameSync(tmp, file);
  } finally {
    try { fs.rmSync(tmp, { force: true }); } catch {}
  }
  fs.chmodSync(file, 0o600);
}

export function getToken() {
  return process.env.MEGABRAIN_TOKEN || loadConfig().token || null;
}

export function setToken(token) {
  saveConfig({ token });
}

export function getEndpoint() {
  const envEndpoint = process.env.MEGABRAIN_ENDPOINT;
  if (typeof envEndpoint === "string" && envEndpoint.trim()) return validateEndpoint(envEndpoint);
  const stored = loadConfig().endpoint;
  if (stored !== undefined && stored !== null) {
    return validateEndpoint(stored);
  }
  return DEFAULT_ENDPOINT;
}

export function setEndpoint(endpoint) {
  saveConfig({ endpoint: validateEndpoint(endpoint) });
}

function runCliAsync(args, envOverrides = {}) {
  return new Promise((resolve) => {
    const env = { ...process.env, ...envOverrides };
    const child = spawn(process.execPath, [binPath, ...args], { env });
    let stdout = '';
    let stderr = '';
    child.stdout.on('data', d => { stdout += d; });
    child.stderr.on('data', d => { stderr += d; });
    child.on('close', (code) => {
      resolve({ status: code, stdout, stderr });
    });
  });
}

test('3. Fallback e prioridade de variaveis de ambiente MEGABRAIN_TOKEN e MEGABRAIN_ENDPOINT', async () => {
  const tmpDir = path.join(os.tmpdir(), `mb-prio-${Date.now()}`);
  const cDir = path.join(tmpDir, 'megabrain');
  fs.mkdirSync(cDir, { recursive: true });

  let lastAuth = null;
  const server = http.createServer((req, res) => {
    lastAuth = req.headers.authorization || null;
    if (req.url === '/health') {
      res.writeHead(200, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify({ status: 'HEALTHY' }));
      return;
    }
    let body = '';
    req.on('data', chunk => { body += chunk; });
    req.on('end', () => {
      const payload = body ? JSON.parse(body) : {};
      res.writeHead(200, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify({ jsonrpc: '2.0', id: payload.id || 1, result: { auth: lastAuth } }));
    });
  });

  await new Promise(r => server.listen(0, '127.0.0.1', r));
  const port = server.address().port;
  const mockEndpoint = `http://127.0.0.1:${port}/mcp`;

  const fileEndpoint = mockEndpoint;
  const fileToken = 'token-from-credentials-file';

  fs.writeFileSync(
    path.join(cDir, 'credentials.json'),
    JSON.stringify({ endpoint: fileEndpoint, token: fileToken }, null, 2),
    { mode: 0o600 }
  );

  const envDefaults = {
    XDG_CONFIG_HOME: tmpDir,
    MEGABRAIN_ENDPOINT: '',
    MEGABRAIN_TOKEN: '',
  };

  try {
    // 3a. Prioridade de Endpoint:
    // Caso 1: Apenas arquivo configurado -> deve usar fileEndpoint
    const s1 = await runCliAsync(['status', '--json'], {
      ...envDefaults,
      MEGABRAIN_ENDPOINT: '',
      MEGABRAIN_TOKEN: '',
    });
    assert.equal(s1.status, 0);
    const p1 = JSON.parse(s1.stdout || '{}');
    assert.equal(p1.endpoint, fileEndpoint, 'Fallback to credentials.json endpoint when no env or flag');

    // Caso 2: Env var setada -> deve sobrepor arquivo
    const envEndpoint = `http://127.0.0.1:${port}/env-mcp`;
    const s2 = await runCliAsync(['status', '--json'], {
      ...envDefaults,
      MEGABRAIN_ENDPOINT: envEndpoint,
    });
    assert.equal(s2.status, 0);
    const p2 = JSON.parse(s2.stdout || '{}');
    assert.equal(p2.endpoint, envEndpoint, 'MEGABRAIN_ENDPOINT must override credentials.json');

    // Caso 3: Flag --endpoint -> deve sobrepor Env var e arquivo
    const flagEndpoint = `http://127.0.0.1:${port}/flag-mcp`;
    const s3 = await runCliAsync(['status', '--json', '--endpoint', flagEndpoint], {
      ...envDefaults,
      MEGABRAIN_ENDPOINT: envEndpoint,
    });
    assert.equal(s3.status, 0);
    const p3 = JSON.parse(s3.stdout || '{}');
    assert.equal(p3.endpoint, flagEndpoint, '--endpoint flag must override MEGABRAIN_ENDPOINT and credentials.json');

    // 3b. Prioridade de Token:
    // Caso 1: Apenas arquivo configurado -> deve enviar fileToken
    lastAuth = null;
    const c1 = await runCliAsync(['call', 'atlas.health', '{}', '--endpoint', mockEndpoint, '--json'], {
      ...envDefaults,
      MEGABRAIN_TOKEN: '',
    });
    assert.equal(c1.status, 0);
    assert.equal(lastAuth, `Bearer ${fileToken}`, 'Fallback to credentials.json token');

    // Caso 2: Env var setada -> deve sobrepor arquivo
    lastAuth = null;
    const c2 = await runCliAsync(['call', 'atlas.health', '{}', '--endpoint', mockEndpoint, '--json'], {
      ...envDefaults,
      MEGABRAIN_TOKEN: 'token-from-env',
    });
    assert.equal(c2.status, 0);
    assert.equal(lastAuth, 'Bearer token-from-env', 'MEGABRAIN_TOKEN must override credentials.json token');

    // Caso 3: Flag --token rejeitada
    const c3 = await runCliAsync(['call', 'test', '{}', '--endpoint', mockEndpoint, '--token', 'token-from-flag', '--json'], {

    assert.equal(l1.status, 0);

    assert.equal(credContent.token, 'another-token');
    assert.equal(credContent.endpoint, fileEndpoint, 'login command without --endpoint must NOT clobber custom endpoint');
    const noTok = await runCliAsync(['login'], { ...envDefaults, MEGABRAIN_TOKEN: '', XDG_CONFIG_HOME: path.join(tmpDir, 'nc') });
    assert.ok(noTok.status === 1 && !fs.existsSync(path.join(tmpDir, 'nc')));
  } finally {
    server.close();
    fs.rmSync(tmpDir, { recursive: true, force: true });
  }
});

async function runWith(result) {
  const lines = [];
  const originalLog = console.log;
  const originalExitCode = process.exitCode;
  console.log = (...parts) => {
    lines.push(parts.map((p) => (typeof p === "string" ? p : JSON.stringify(p))).join(" "));
  };
  process.exitCode = undefined;
  let failure = null;
  try {
    await run({ positionals: ["atlas.health", "{}"], flags: { json: true }, callToolFn: async () => result });
  } catch (e) {
    failure = e;
  } finally {

  process.exitCode = originalExitCode;
  if (failure) throw failure;
  return { exitCode, stdout: lines.join("\n") };
}
